cmake_minimum_required(VERSION 3.15)
project(aora_connect_portable LANGUAGES CXX RC)
if(NOT EXISTS "${PAYLOAD_ZIP}")
  message(FATAL_ERROR "PAYLOAD_ZIP must point to the packaged application ZIP")
endif()
file(TO_CMAKE_PATH "${PAYLOAD_ZIP}" PAYLOAD_ZIP)
configure_file(payload.rc.in payload.rc @ONLY)
add_executable(AORAConnect WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_compile_features(AORAConnect PRIVATE cxx_std_17)
target_compile_definitions(AORAConnect PRIVATE UNICODE _UNICODE PACKAGE_ID=L"${PACKAGE_ID}")
set_property(TARGET AORAConnect PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(AORAConnect PRIVATE shell32 ole32 crypt32)
target_link_options(AORAConnect PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
